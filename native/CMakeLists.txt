cmake_minimum_required(VERSION 3.24)
project(SemiInspectX LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 4.14.0 EXACT REQUIRED COMPONENTS core imgproc imgcodecs)
add_library(semiinspect SHARED src/engine.cpp)
target_compile_definitions(semiinspect PRIVATE SI_BUILD)
target_include_directories(semiinspect PUBLIC include PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(semiinspect PRIVATE ${OpenCV_LIBS})
add_executable(si_generate src/generate.cpp)
target_include_directories(si_generate PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(si_generate PRIVATE ${OpenCV_LIBS})
include(CTest)
if(BUILD_TESTING)
  include(FetchContent)
  set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
  FetchContent_Declare(googletest URL https://codeload.github.com/google/googletest/zip/refs/tags/v1.17.0
    URL_HASH SHA256=40d4ec942217dcc84a9ebe2a68584ada7d4a33a8ee958755763278ea1c5e18ff)
  FetchContent_MakeAvailable(googletest)
  add_executable(si_tests tests/engine_tests.cpp)
  target_include_directories(si_tests PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(si_tests PRIVATE semiinspect GTest::gtest_main ${OpenCV_LIBS})
  include(GoogleTest)
  gtest_discover_tests(si_tests DISCOVERY_MODE PRE_TEST)
endif()
if(MSVC)
  target_compile_options(semiinspect PRIVATE /W4 /WX)
  target_compile_options(si_generate PRIVATE /W4 /WX)
endif()
